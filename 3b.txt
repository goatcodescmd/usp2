#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <sys/stat.h>
#include <pwd.h>
#include <grp.h>
#include <time.h>

void print_permissions(mode_t mode) {
    printf((S_ISDIR(mode))  ? "d" : "-");
    printf((mode & S_IRUSR) ? "r" : "-");
    printf((mode & S_IWUSR) ? "w" : "-");
    printf((mode & S_IXUSR) ? "x" : "-");
    printf((mode & S_IRGRP) ? "r" : "-");
    printf((mode & S_IWGRP) ? "w" : "-");
    printf((mode & S_IXGRP) ? "x" : "-");
    printf((mode & S_IROTH) ? "r" : "-");
    printf((mode & S_IWOTH) ? "w" : "-");
    printf((mode & S_IXOTH) ? "x" : "-");
}

int main(int argc, char *argv[]) {
    DIR *dp;
    struct dirent *entry;
    struct stat sb;
    char filepath[1024];
    const char *dirpath;
    struct passwd *pw;
    struct group *gr;
    char timebuf[64];

    dirpath = (argc > 1) ? argv[1] : ".";

    dp = opendir(dirpath);
    if (dp == NULL) {
        perror("opendir");
        return 1;
    }

    printf("%-10s %-6s %-10s %-10s %-10s %-20s %s\n",
           "Inode", "Links", "Owner", "Group", "Size", "Modified", "Name");
    printf("%-10s %-6s %-10s %-10s %-10s %-20s %s\n",
           "------", "-----", "-----", "-----", "----", "--------", "----");

    while ((entry = readdir(dp)) != NULL) {
        if (strcmp(entry->d_name, ".") == 0 || strcmp(entry->d_name, "..") == 0)
            continue;

        snprintf(filepath, sizeof(filepath), "%s/%s", dirpath, entry->d_name);

        if (lstat(filepath, &sb) < 0) {
            perror("lstat");
            continue;
        }

        pw = getpwuid(sb.st_uid);
        gr = getgrgid(sb.st_gid);
        strftime(timebuf, sizeof(timebuf), "%Y-%m-%d %H:%M", localtime(&sb.st_mtime));

        // 1. Inode
        printf("%-10ld ", (long)sb.st_ino);
        // 2. Permissions
        print_permissions(sb.st_mode);
        // 3. Hard links
        printf(" %-4ld ", (long)sb.st_nlink);
        // 4. Owner
        printf("%-10s ", pw ? pw->pw_name : "?");
        // 5. Size
        printf("%-10ld ", (long)sb.st_size);
        // 6. Modification time
        printf("%-20s ", timebuf);
        // 7. Name
        printf("%s\n", entry->d_name);
    }

    closedir(dp);
    return 0;
}
