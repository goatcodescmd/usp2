// ===================== FILE 2: 12b.c =====================
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/types.h>

int main() {
    pid_t pid;
    int status;

    printf("Main PID: %d\n\n", getpid());

    pid = fork();
    if (pid < 0) { perror("fork"); return 1; }

    if (pid == 0) {
        // Child: set user-defined env vars, then exec
        printf("[Child] PID: %d\n", getpid());
        setenv("MY_VAR1", "HelloFromChild", 1);
        setenv("MY_VAR2", "UnixLab2026", 1);
        setenv("CHILD_MSG", "I am the child process", 1);

        printf("[Child] Executing envprint in child mode (user-defined vars):\n");
        execl("./envprint", "envprint", "child", (char *)NULL);
        perror("execl");
        _exit(1);
    } else {
        // Parent: wait for child, then exec to show default vars
        waitpid(pid, &status, 0);
        printf("\n[Parent] PID: %d\n", getpid());
        printf("[Parent] Executing envprint in parent mode (default env vars):\n");
        execl("./envprint", "envprint", "parent", (char *)NULL);
        perror("execl");
    }

    return 0;
}
