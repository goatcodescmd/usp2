// ===================== FILE 2: myinterp (interpreter file) =====================
// Create the interpreter script:
// vi myinterp
// Make it executable: chmod +x myinterp
// Content of myinterp:
/*
#!/home/user/echoall ARG_FROM_INTERP
*/
// NOTE: Replace /home/user/ with the actual full path where echoall is compiled.
// Use: pwd   to find your current directory

// ===================== FILE 3: 4b.c (main program) =====================
// vi 4b.c

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <sys/types.h>

int main() {
    pid_t pid;

    pid = fork();
    if (pid < 0) {
        perror("fork");
        return 1;
    }

    if (pid == 0) {
        // Child: execute the interpreter file with arguments
        printf("Child PID: %d\n", getpid());
        printf("Child executing interpreter file...\n");
        // Pass arguments to the interpreter file
        execl("./myinterp", "myinterp", "arg1_from_child", "arg2_from_child", (char *)NULL);
        perror("execl");
        _exit(1);
    } else {
        // Parent waits
        printf("Parent PID: %d, waiting for child...\n", getpid());
        wait(NULL);
        printf("Child finished.\n");
    }

    return 0;
}

/*
HOW TO RUN (in order):
1. vi echoall.c         (write echoall.c as shown above, then :wq)

2. vi myinterp          (write: #!/full/path/to/echoall ARG_FROM_INTERP)

   (use 'pwd' to get full path for the shebang line)
