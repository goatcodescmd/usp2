#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/types.h>
#include <sys/wait.h>
#include <signal.h>

// ===== PART 1: Demonstrate Zombie Process =====

void demo_zombie() {
    pid_t pid = fork();
    if (pid < 0) { perror("fork"); return; }

    if (pid == 0) {
        // Child exits immediately
        printf("[Child] PID %d exiting (will become zombie if parent doesn't wait)\n", getpid());
        exit(0);
    } else {
        // Parent does NOT call wait() - child becomes zombie
        printf("[Parent] PID %d - NOT calling wait(). Child %d is now a zombie.\n", getpid(), pid);
        printf("[Parent] Check with: ps -aux | grep defunct\n");
        sleep(10);  // During this sleep, child is zombie
        printf("[Parent] Now calling wait() to reap zombie...\n");
        wait(NULL);
        printf("[Parent] Zombie reaped.\n");
    }
}

// ===== PART 2: Solution using SIGCHLD signal handler =====

void sigchld_handler(int sig) {
    pid_t pid;
    int status;
    // Non-blocking wait for all terminated children
    while ((pid = waitpid(-1, &status, WNOHANG)) > 0) {
        printf("[SIGCHLD Handler] Reaped zombie child PID: %d\n", pid);
    }
}

void demo_solution() {
    pid_t pid;

    // Install SIGCHLD handler to auto-reap children
    signal(SIGCHLD, sigchld_handler);

    pid = fork();
    if (pid < 0) { perror("fork"); return; }

    if (pid == 0) {
        printf("[Child] PID %d exiting (parent has SIGCHLD handler)\n", getpid());
        exit(0);
    } else {
        printf("[Parent] PID %d - SIGCHLD handler installed, no zombie!\n", getpid());
        sleep(3);  // Handler auto-reaps child during this sleep
        printf("[Parent] Done. No zombie was created.\n");
    }
}

int main() {
    printf("===== Demo 1: Zombie Process =====\n");
    demo_zombie();

    printf("\n===== Demo 2: Solution (SIGCHLD handler) =====\n");
    demo_solution();

    return 0;
}
