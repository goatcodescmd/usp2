#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <string.h>

void display_chars(int fd, int count, const char *label) {
    char buf[256];
    ssize_t n = read(fd, buf, count);
    if (n < 0) { perror("read"); return; }
    buf[n] = '\0';
    printf("%s [%ld chars]: %s\n", label, (long)n, buf);
}

int main(int argc, char *argv[]) {
    int fd;
    struct stat sb;
    const char *filename;

    filename = (argc > 1) ? argv[1] : "seektest.txt";

    fd = open(filename, O_RDONLY);
    if (fd < 0) { perror("open"); return 1; }

    // i. Read first 20 characters
    lseek(fd, 0, SEEK_SET);
    display_chars(fd, 20, "i.  First 20 chars");

    // ii. Seek to 10th byte from beginning, display 20 chars
    lseek(fd, 10, SEEK_SET);
    printf("    (seeked to byte 10 from start)\n");
    display_chars(fd, 20, "ii. 20 chars from byte 10");

    // iii. Seek 10 bytes ahead from current offset, display 20 chars
    lseek(fd, 10, SEEK_CUR);
    printf("    (seeked 10 bytes ahead from current)\n");
    display_chars(fd, 20, "iii.20 chars from there");

    // iv. Display file size
    fstat(fd, &sb);
    printf("\niv. File size: %ld bytes\n", (long)sb.st_size);

    close(fd);
    return 0;
}
