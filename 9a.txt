#include <stdio.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <fcntl.h>
#include <unistd.h>

void print_perms(const char *filename) {
    struct stat sb;
    stat(filename, &sb);
    printf("Permissions of %s: ", filename);
    printf((sb.st_mode & S_IRUSR) ? "r" : "-");
    printf((sb.st_mode & S_IWUSR) ? "w" : "-");
    printf((sb.st_mode & S_IXUSR) ? "x" : "-");
    printf((sb.st_mode & S_IRGRP) ? "r" : "-");
    printf((sb.st_mode & S_IWGRP) ? "w" : "-");
    printf((sb.st_mode & S_IXGRP) ? "x" : "-");
    printf((sb.st_mode & S_IROTH) ? "r" : "-");
    printf((sb.st_mode & S_IWOTH) ? "w" : "-");
    printf((sb.st_mode & S_IXOTH) ? "x" : "-");
    printf(" (%o)\n", (unsigned)(sb.st_mode & 0777));
}

int main() {
    int fd;
    mode_t old_mask;

    // --- Demonstrate umask ---
    printf("=== umask Demonstration ===\n");
    old_mask = umask(0);
    printf("Old umask: %03o\n", (unsigned)old_mask);

    // Create file with umask 0 (no masking)
    fd = open("file_umask0.txt", O_CREAT | O_WRONLY, 0777);
    if (fd >= 0) { close(fd); print_perms("file_umask0.txt"); }

    // Set umask to 022 (common default)
    umask(022);
    printf("Set umask to 022\n");
    fd = open("file_umask022.txt", O_CREAT | O_WRONLY, 0777);
    if (fd >= 0) { close(fd); print_perms("file_umask022.txt"); }

    // Set umask to 077 (very restrictive)
    umask(077);
    printf("Set umask to 077\n");
    fd = open("file_umask077.txt", O_CREAT | O_WRONLY, 0777);
    if (fd >= 0) { close(fd); print_perms("file_umask077.txt"); }

    printf("\n=== chmod Demonstration ===\n");
    // Change permissions of file using chmod
    chmod("file_umask0.txt", 0644);
    print_perms("file_umask0.txt");

    chmod("file_umask0.txt", 0755);
    print_perms("file_umask0.txt");

    chmod("file_umask0.txt", 0400);
    print_perms("file_umask0.txt");

    // Restore old mask
    umask(old_mask);
    printf("\nRestored old umask: %03o\n", (unsigned)old_mask);

    return 0;
}
