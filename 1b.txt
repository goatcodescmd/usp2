#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    int fd;
    char buf[10];
    pid_t pid;

    fd = open("testfile.txt", O_RDONLY);
    if (fd < 0) {
        perror("open");
        return 1;
    }

    // Parent reads 5 bytes, offset moves to 5
    read(fd, buf, 5);
    buf[5] = '\0';
    printf("Parent read: %s\n", buf);
    printf("Parent offset after read: %ld\n", (long)lseek(fd, 0, SEEK_CUR));

    pid = fork();
    if (pid < 0) {
        perror("fork");
        return 1;
    }

    if (pid == 0) {
        // Child process - shares same file offset
        printf("Child offset (shared with parent): %ld\n", (long)lseek(fd, 0, SEEK_CUR));
        read(fd, buf, 5);
        buf[5] = '\0';
        printf("Child read (continues from parent's offset): %s\n", buf);
        close(fd);
        _exit(0);
    } else {
        wait(NULL);
        printf("Parent: Both parent and child share the SAME file offset (inherited via fork)\n");
        close(fd);
    }

    return 0;
}
