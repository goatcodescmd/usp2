#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
#include <errno.h>
#include <string.h>
#include <sys/stat.h>

#define BUFSIZE 4096

int main(int argc, char *argv[]) {
    int fd_src, fd_dst;
    char buf[BUFSIZE];
    ssize_t n_read, n_written;
    struct stat sb;

    if (argc != 3) {
        fprintf(stderr, "Usage: %s <source> <destination>\n", argv[0]);
        return 1;
    }

    // Open source
    fd_src = open(argv[1], O_RDONLY);
    if (fd_src < 0) {
        fprintf(stderr, "Error opening source '%s': %s\n", argv[1], strerror(errno));
        return 1;
    }

    // Check source is not a directory
    if (fstat(fd_src, &sb) < 0) {
        perror("fstat");
        close(fd_src);
        return 1;
    }
    if (S_ISDIR(sb.st_mode)) {
        fprintf(stderr, "Error: '%s' is a directory\n", argv[1]);
        close(fd_src);
        return 1;
    }

    // Open destination (create/truncate)
    fd_dst = open(argv[2], O_WRONLY | O_CREAT | O_TRUNC, sb.st_mode & 0777);
    if (fd_dst < 0) {
        fprintf(stderr, "Error opening destination '%s': %s\n", argv[2], strerror(errno));
        close(fd_src);
        return 1;
    }

    // Copy loop
    while ((n_read = read(fd_src, buf, BUFSIZE)) > 0) {
        n_written = write(fd_dst, buf, n_read);
        if (n_written != n_read) {
            fprintf(stderr, "Write error: %s\n", strerror(errno));
            close(fd_src);
            close(fd_dst);
            return 1;
        }
    }

    if (n_read < 0) {
        fprintf(stderr, "Read error: %s\n", strerror(errno));
        close(fd_src);
        close(fd_dst);
        return 1;
    }

    printf("File '%s' copied to '%s' successfully.\n", argv[1], argv[2]);

    close(fd_src);
    close(fd_dst);
    return 0;
}
