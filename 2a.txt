#include <stdio.h>
#include <sys/stat.h>
#include <time.h>
#include <pwd.h>
#include <grp.h>

int main(int argc, char *argv[]) {
    struct stat sb;
    struct passwd *pw;
    struct group *gr;

    if (argc < 2) {
        printf("Usage: %s <filename>\n", argv[0]);
        return 1;
    }

    if (stat(argv[1], &sb) < 0) {
        perror("stat");
        return 1;
    }

    printf("File Statistics for: %s\n", argv[1]);
    printf("----------------------------------\n");
    printf("1. Inode number     : %ld\n", (long)sb.st_ino);
    printf("2. File size        : %ld bytes\n", (long)sb.st_size);
    printf("3. Number of links  : %ld\n", (long)sb.st_nlink);
    printf("4. Block size       : %ld bytes\n", (long)sb.st_blksize);
    printf("5. Blocks allocated : %ld\n", (long)sb.st_blocks);

    pw = getpwuid(sb.st_uid);
    printf("6. Owner            : %s (UID: %d)\n", pw ? pw->pw_name : "unknown", sb.st_uid);

    gr = getgrgid(sb.st_gid);
    printf("7. Group            : %s (GID: %d)\n", gr ? gr->gr_name : "unknown", sb.st_gid);

    printf("8. Last modified    : %s", ctime(&sb.st_mtime));
    printf("9. Last accessed    : %s", ctime(&sb.st_atime));

    printf("10. Permissions     : ");
    printf((S_ISDIR(sb.st_mode)) ? "d" : "-");
    printf((sb.st_mode & S_IRUSR) ? "r" : "-");
    printf((sb.st_mode & S_IWUSR) ? "w" : "-");
    printf((sb.st_mode & S_IXUSR) ? "x" : "-");
    printf((sb.st_mode & S_IRGRP) ? "r" : "-");
    printf((sb.st_mode & S_IWGRP) ? "w" : "-");
    printf((sb.st_mode & S_IXGRP) ? "x" : "-");
    printf((sb.st_mode & S_IROTH) ? "r" : "-");
    printf((sb.st_mode & S_IWOTH) ? "w" : "-");
    printf((sb.st_mode & S_IXOTH) ? "x" : "-");
    printf("\n");

    return 0;
}
